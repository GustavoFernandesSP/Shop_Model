package com.loja.shop.config;

import com.loja.shop.security.JwtAuthFilter;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;
import java.util.List;

@Configuration
public class SecurityConfig {

    @Bean
    SecurityFilterChain filterChain(HttpSecurity http, JwtAuthFilter jwtFilter) throws Exception {
        http
            .csrf(csrf -> csrf.disable())   // cookie SameSite=Lax + CORS restrito; revisar para produÃ§Ã£o
            .cors(Customizer.withDefaults())
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(a -> a
                .requestMatchers(HttpMethod.OPTIONS, "/**").permitAll()
                .requestMatchers("/api/register", "/api/login", "/api/logout", "/api/get-products").permitAll()
                .anyRequest().authenticated())
            // o front sempre faz response.json(), entÃ£o 401 tambÃ©m precisa devolver JSON
            .exceptionHandling(e -> e.authenticationEntryPoint((req, res, ex) -> {
                res.setStatus(401);
                res.setContentType("application/json;charset=UTF-8");
                res.getWriter().write("{\"error\":\"Unauthorized\"}");
            }))
            .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }

    @Bean
    PasswordEncoder passwordEncoder() { return new BCryptPasswordEncoder(); }

    @Bean
    CorsConfigurationSource corsConfigurationSource(
            @Value("${app.cors.allowed-origins}") List<String> origins) {
        CorsConfiguration cfg = new CorsConfiguration();
        cfg.setAllowedOrigins(origins);                 // com credentials nÃ£o pode ser "*"
        cfg.setAllowedMethods(List.of("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"));
        cfg.setAllowedHeaders(List.of("*"));
        cfg.setAllowCredentials(true);                  // front usa credentials: "include"
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", cfg);
        return source;
    }
}

package com.loja.shop.controller;

import com.loja.shop.dto.Requests;
import com.loja.shop.model.AppUser;
import com.loja.shop.repository.UserRepository;
import com.loja.shop.security.JwtAuthFilter;
import com.loja.shop.security.JwtService;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpHeaders;
import org.springframework.http.ResponseCookie;
import org.springframework.http.ResponseEntity;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.web.bind.annotation.*;
import java.time.Duration;
import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/api")
@RequiredArgsConstructor
public class AuthController {
    private final UserRepository users;
    private final PasswordEncoder encoder;
    private final JwtService jwt;

    @PostMapping("/register")
    public ResponseEntity<Map<String, Object>> register(@RequestBody Requests.Register req) {
        if (req.email() == null || req.email().isBlank() || req.password() == null || req.password().length() < 6) {
            return ApiResponse.error(400, "Informe e-mail e uma senha com no mÃ­nimo 6 caracteres");
        }
        String email = req.email().trim().toLowerCase();
        if (users.existsByEmail(email)) {
            return ApiResponse.error(409, "Este e-mail jÃ¡ estÃ¡ cadastrado");
        }
        AppUser u = new AppUser();
        u.setName(req.username() != null && !req.username().isBlank() ? req.username() : req.name());
        u.setEmail(email);
        u.setPasswordHash(encoder.encode(req.password()));
        users.save(u);
        return loggedIn(u);
    }

    @PostMapping("/login")
    public ResponseEntity<Map<String, Object>> login(@RequestBody Requests.Login req) {
        if (req.email() == null || req.password() == null) {
            return ApiResponse.error(400, "Informe e-mail e senha");
        }
        AppUser u = users.findByEmail(req.email().trim().toLowerCase()).orElse(null);
        if (u == null || !encoder.matches(req.password(), u.getPasswordHash())) {
            return ApiResponse.error(401, "E-mail ou senha invÃ¡lidos");
        }
        return loggedIn(u);
    }

    // o front chama /logout com GET
    @GetMapping("/logout")
    public ResponseEntity<Map<String, Object>> logout() {
        return ResponseEntity.ok()
                .header(HttpHeaders.SET_COOKIE, cookie("", Duration.ZERO).toString())
                .body(Map.of("message", "Logout realizado"));
    }

    private ResponseEntity<Map<String, Object>> loggedIn(AppUser u) {
        Map<String, Object> user = new HashMap<>();
        user.put("_id", u.getId());   // o front pode ler _id ou id
        user.put("id", u.getId());
        user.put("name", u.getName());
        user.put("username", u.getName());
        user.put("email", u.getEmail());
        return ResponseEntity.ok()
                .header(HttpHeaders.SET_COOKIE, cookie(jwt.create(u.getId()), Duration.ofDays(7)).toString())
                .body(Map.of("user", user));
    }

    private ResponseCookie cookie(String value, Duration maxAge) {
        return ResponseCookie.from(JwtAuthFilter.COOKIE, value)
                .httpOnly(true)
                .secure(false)        // em produÃ§Ã£o com HTTPS: true
                .sameSite("Lax")
                .path("/")
                .maxAge(maxAge)
                .build();
    }
}

package com.loja.shop.controller;

import com.loja.shop.dto.Requests;
import com.loja.shop.model.*;
import com.loja.shop.repository.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.bind.annotation.*;
import java.math.BigDecimal;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.time.temporal.ChronoUnit;

@RestController
@RequestMapping("/api")
@RequiredArgsConstructor
public class OrderController {
    private final OrderRepository orders;
    private final ProductRepository products;
    private final UserRepository users;

    @PostMapping("/place-order")
    @Transactional
    public ResponseEntity<Map<String, Object>> placeOrder(@AuthenticationPrincipal Long userId,
                                                          @RequestBody Requests.PlaceOrder req) {
        AppUser user = users.findById(userId).orElse(null);
        if (user == null) return ApiResponse.error(401, "Unauthorized");
        if (req.items() == null || req.items().isEmpty()) return ApiResponse.error(400, "Carrinho vazio");

        CustomerOrder o = new CustomerOrder();
        o.setUser(user);   // usuÃ¡rio vem do token, nÃ£o do user_id do payload
        o.setDeliveryType(req.delivery_type());
        o.setPromoCode(req.promo_code());
        o.setContactNumber(req.contact_number());

        BigDecimal subtotal = BigDecimal.ZERO;
        int count = 0;
        for (Requests.CartItem ci : req.items()) {
            Product p = products.findById(ci.id()).orElse(null);
            if (p == null) return ApiResponse.error(400, "Produto nÃ£o encontrado: " + ci.id());
            int qty = ci.quantity() == null ? 1 : Math.max(1, ci.quantity());

            OrderItem it = new OrderItem();
            it.setOrder(o);
            it.setProductId(p.getId());
            it.setName(p.getName());
            it.setDescription(p.getDescription());
            it.setProductImage(p.getImage());
            it.setPrice(p.getPrice());   // preÃ§o SEMPRE do banco, nunca do cliente
            it.setQuantity(qty);
            o.getItems().add(it);

            subtotal = subtotal.add(p.getPrice().multiply(BigDecimal.valueOf(qty)));
            count += qty;
        }
        // TODO: calcular o frete no servidor (hoje confia no valor enviado pelo front)
        BigDecimal delivery = req.delivery_type_cost() == null ? BigDecimal.ZERO : req.delivery_type_cost();
        o.setDeliveryTypeCost(delivery);
        o.setTotalItemCount(count);
        o.setCostBeforeDelivery(subtotal);
        o.setCostAfterDelivery(subtotal.add(delivery));
        orders.save(o);

        return ResponseEntity.ok(Map.of("message", "Pedido realizado com sucesso!", "order_id", o.getId()));
    }

    @GetMapping("/get-orders/{userId}")
    @Transactional(readOnly = true)
    public ResponseEntity<Map<String, Object>> getOrders(@AuthenticationPrincipal Long me,
                                                         @PathVariable Long userId) {
        if (!me.equals(userId)) return ApiResponse.error(403, "Forbidden");
        List<Map<String, Object>> list = orders.findByUser_IdOrderByCreatedAtDesc(userId)
                .stream().map(this::toDto).toList();
        return ResponseEntity.ok(Map.of("orders", list));
    }

    @PostMapping("/cancel-order")
    @Transactional
    public ResponseEntity<Map<String, Object>> cancelOrder(@AuthenticationPrincipal Long me,
                                                           @RequestBody Requests.CancelOrder req) {
        CustomerOrder o = req.order_id() == null ? null : orders.findById(req.order_id()).orElse(null);
        if (o == null || !o.getUser().getId().equals(me)) return ApiResponse.error(404, "Pedido nÃ£o encontrado");
        if (!"pending".equals(o.getStatus())) return ApiResponse.error(400, "Este pedido nÃ£o pode mais ser cancelado");
        o.setStatus("cancelled");
        return ResponseEntity.ok(Map.of("message", "Pedido cancelado", "user_id", me));
    }

    private Map<String, Object> toDto(CustomerOrder o) {
        String status = o.getStatus();
        int days = "Express".equalsIgnoreCase(o.getDeliveryType()) ? 3 : 7;

        Map<String, Object> m = new LinkedHashMap<>();
        m.put("_id", String.valueOf(o.getId()));   // o front usa _id.slice(...)
        m.put("user_id", o.getUser().getId());
        m.put("status", status);
        m.put("createdAt", o.getCreatedAt().toString());
        m.put("expected_delivery_date", o.getCreatedAt().plus(days, ChronoUnit.DAYS).toString());
        m.put("percentage_complete", percentage(status));
        m.put("order_cancelled", "cancelled".equals(status));
        m.put("order_processed", "delivered".equals(status));
        m.put("totalItemCount", o.getTotalItemCount());
        m.put("delivery_type", o.getDeliveryType());
        m.put("delivery_type_cost", o.getDeliveryTypeCost());
        m.put("cost_before_delivery_rate", o.getCostBeforeDelivery());
        m.put("cost_after_delivery_rate", o.getCostAfterDelivery());
        m.put("promo_code", o.getPromoCode());
        m.put("contact_number", o.getContactNumber());
        m.put("items", o.getItems().stream().map(i -> {
            Map<String, Object> im = new LinkedHashMap<>();
            im.put("_id", i.getProductId());
            im.put("name", i.getName());
            im.put("description", i.getDescription());
            im.put("product_image", i.getProductImage());
            im.put("price", i.getPrice());
            im.put("quantity", i.getQuantity());
            return im;
        }).toList());
        return m;
    }

    // a barra de progresso e as faixas do front dependem destes valores
    private int percentage(String status) {
        return switch (String.valueOf(status)) {
            case "paid" -> 40;
            case "processing" -> 60;
            case "shipped" -> 75;
            case "delivered" -> 100;
            case "cancelled" -> 0;
            default -> 10;   // pending
        };
    }
}

package com.loja.shop.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.Cookie;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import java.io.IOException;
import java.util.List;

@Component
@RequiredArgsConstructor
public class JwtAuthFilter extends OncePerRequestFilter {
    public static final String COOKIE = "token";
    private final JwtService jwt;

    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
            throws ServletException, IOException {
        if (req.getCookies() != null) {
            for (Cookie c : req.getCookies()) {
                if (COOKIE.equals(c.getName())) {
                    try {
                        Long userId = jwt.parseUserId(c.getValue());
                        SecurityContextHolder.getContext().setAuthentication(
                                new UsernamePasswordAuthenticationToken(userId, null, List.of()));
                    } catch (Exception ignored) { /* token invÃ¡lido/expirado: segue sem autenticaÃ§Ã£o */ }
                }
            }
        }
        chain.doFilter(req, res);
    }
}

package com.loja.shop.security;

import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;

@Service
public class JwtService {
    private static final long DAYS_7 = 7L * 24 * 60 * 60 * 1000;
    private final SecretKey key;

    public JwtService(@Value("${app.jwt.secret}") String secret) {
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8)); // mÃ­nimo 32 caracteres
    }

    public String create(Long userId) {
        Date now = new Date();
        return Jwts.builder()
                .subject(String.valueOf(userId))
                .issuedAt(now)
                .expiration(new Date(now.getTime() + DAYS_7))
                .signWith(key)
                .compact();
    }

    public Long parseUserId(String token) {
        String sub = Jwts.parser().verifyWith(key).build()
                .parseSignedClaims(token).getPayload().getSubject();
        return Long.valueOf(sub);
    }
}
